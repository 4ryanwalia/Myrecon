# Editorial fragments that get inlined into the generated breach articles.
# They have no <head>, no heading and no navigation, and their text already
# appears in full on the article that includes them. These also 301 to
# /breaches/ at the edge; the Disallow saves the crawl entirely.
Disallow: /content/

# variants are handled by the self-referencing canonical on every page.
